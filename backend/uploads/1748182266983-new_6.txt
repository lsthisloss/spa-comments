import { useCallback, useEffect, useRef } from "react";
import { observer } from "mobx-react-lite";
import { postStore } from "../../services/PostStore";
import { Empty, Button, Badge } from "antd";
import { useNavigate } from "react-router-dom";
import { userStore } from "../../services/UserStore";
import { socketStore } from "../../services/SocketStore";
import { ArrowUpOutlined } from "@ant-design/icons";

import { useVirtualPosts } from "../../hooks/useVirtualPosts";
import { usePostsLoader } from "../../hooks/usePostsLoader";
import { LoadingIndicator } from "../ui/LoadingIndicator";
import { DebugInfo } from "../ui/DebugInfo";
import { NewPostNotification } from "../ui/NewPostNotification";
import { MemoizedPostItem } from "../ui/MemoizedPostItem";

const AUTO_LOAD_SCROLL_THRESHOLD = 300;
const LOAD_MORE_THRESHOLD = 5;

interface PostsFeedProps {
  activeTab: string;
}

const PostsThread = observer(({ activeTab }: PostsFeedProps) => {
  const navigate = useNavigate();

  const isFeed = activeTab === "all";
  const postList = isFeed ? postStore.feedList : postStore.followingList;
  const loading = isFeed ? postStore.feedLoading : postStore.followingLoading;
  const allLoaded = isFeed ? postStore.feedAllLoaded : postStore.followingAllLoaded;
  const manualUpdateMode = isFeed ? postStore.feedManualUpdateMode : postStore.followingManualUpdateMode;
  const newPostsCount = isFeed ? postStore.feedNewPostsCount : postStore.followingNewPostsCount;
  const latestPost = isFeed ? postStore.feedLatestPost : postStore.followingLatestPost;
  const totalPosts = isFeed ? postStore.feedTotal : postStore.followingTotal;
  
  const { loadMorePosts } = usePostsLoader(isFeed, loading, allLoaded);
  
  const { 
    virtualItems, 
    totalHeight,
    scrollTop,
    measureElement,
  } = useVirtualPosts(postList);

  // Проверка, находимся ли мы в верхней части списка
  const isAtTop = scrollTop <= AUTO_LOAD_SCROLL_THRESHOLD;

  // Подгрузка при приближении к концу
  useEffect(() => {
    if (loading || allLoaded || virtualItems.length === 0 || totalPosts === 0) return;

    const lastItem = virtualItems[virtualItems.length - 1];
    const remainingItems = postList.length - lastItem.index;
    
    if (remainingItems <= LOAD_MORE_THRESHOLD) {
      loadMorePosts();
    }
  }, [virtualItems, postList.length, loading, allLoaded, loadMorePosts, totalPosts]);

  // Подгрузка по скроллу страницы
  useEffect(() => {
    const handleWindowScroll = () => {
      if (loading || allLoaded || totalPosts === 0) return;

      const scrollHeight = document.documentElement.scrollHeight;
      const scrollTop = window.scrollY;
      const clientHeight = window.innerHeight;
      
      if (scrollTop + clientHeight >= scrollHeight * 0.8) {
        loadMorePosts();
      }
    };

    const throttledHandleScroll = throttle(handleWindowScroll, 300);
    window.addEventListener('scroll', throttledHandleScroll, { passive: true });
    
    return () => {
      window.removeEventListener('scroll', throttledHandleScroll);
    };
  }, [loading, allLoaded, loadMorePosts, totalPosts]);

  const handleLoadNewPosts = useCallback(() => {
    if (isFeed) {
      postStore.handleLoadNewFeedPosts();
    } else {
      postStore.handleLoadNewFollowingPosts();
    }

    // Прокручиваем браузер в начало
    window.scrollTo({ top: 0, behavior: 'auto' });
  }, [isFeed]);

  // Загрузка данных при первом рендере
  useEffect(() => {
    if (postList.length === 0 && !loading ) {
      if (isFeed) {
        postStore.fetchFeedPosts(1);
      } else if (userStore.user?.id) {
        postStore.fetchFollowingPosts(userStore.user.id, 1);
      }
    }
  }, [isFeed, loading, postList.length, totalPosts]);

  // Автоматическое обновление, если мы в верхней части страницы
  useEffect(() => {
    if (isAtTop && !manualUpdateMode && newPostsCount > 0) {
      handleLoadNewPosts();
    }
  }, [isAtTop, manualUpdateMode, newPostsCount, handleLoadNewPosts]);

  // Убедимся, что сокет-слушатели настроены
  useEffect(() => {
    // PostStore автоматически настраивает слушателей сокетов в конструкторе,
    // нам просто нужно убедиться, что socketStore инициализирован
    if (!socketStore.posts) {
      console.warn("Socket connection is not established");
      return;
    }

    // При размонтировании компонента мы не отключаем слушателей,
    // так как они нужны для работы всего приложения
    return () => {
      // Ничего не делаем при размонтировании
    };
  }, []);

  // При переключении между feed и following, включаем мануальный режим
  // для предотвращения внезапного появления постов
  useEffect(() => {
    if (isFeed) {
      postStore.setFeedManualUpdateMode(true);
    } else {
      postStore.setFollowingManualUpdateMode(true);
    }
  }, [isFeed]);

  const handlePostClick = useCallback((postId: string) => {
    postStore.saveFeedState(window.scrollY);
    navigate(`/post/${postId}`);
  }, [navigate]);

  const handleFocusPost = useCallback((postId: string) => {
    handleLoadNewPosts();

    setTimeout(() => {
      const postIndex = postList.findIndex(post => post.id === postId);
      if (postIndex >= 0) {
        // Прокручиваем к посту используя его позицию
        const postElement = document.getElementById(`post-${postList[postIndex].id}`);
        if (postElement) {
          postElement.scrollIntoView({ behavior: 'smooth', block: 'start' });
        }
      }
    }, 500);
  }, [handleLoadNewPosts, postList]);

  const truncateContent = useCallback((text: string, maxLength = 50) => {
    if (!text) return '';
    return text.length > maxLength ? `${text.substring(0, maxLength)}...` : text;
  }, []);

  // Проверяем есть ли у нас данные для отображения
  if (totalPosts === 0 && !loading) {
    return <Empty description="You have no more posts" style={{ margin: "60px auto" }} />;
  }

  if (!loading && postList.length === 0) {
    return <Empty description="You have no posts anymore" style={{ margin: "60px auto" }} />;
  }

  return (
    <div>
      {manualUpdateMode && newPostsCount > 0 && (
        <div style={{ 
          position: 'sticky', 
          top: 0, 
          zIndex: 999, 
          backgroundColor: '#fff', 
          padding: '10px 0',
          textAlign: 'center',
          borderBottom: '1px solid #f0f0f0'
        }}>
          {latestPost ? (
            <NewPostNotification 
              latestPost={latestPost}
              onFocusPost={handleFocusPost}
              newPostsCount={newPostsCount}
              onLoadNewPosts={handleLoadNewPosts}
              truncateContent={truncateContent}
            />
          ) : (
            <Button 
              type="primary" 
              icon={<ArrowUpOutlined />}
              onClick={handleLoadNewPosts}
            >
              <Badge count={newPostsCount} size="small" offset={[5, -3]}>
                Show new posts
              </Badge>
            </Button>
          )}
        </div>
      )}

      {/* Основной контейнер */}
      <div
        style={{
          width: '100%',
          position: 'relative',
          height: `${totalHeight}px`,
        }}
      >
        {virtualItems.map((virtualItem) => (
          <div
            key={virtualItem.key}
            id={`post-${virtualItem.post.id}`}
            ref={(el) => measureElement(el, virtualItem.index)}
            style={{
              position: 'absolute',
              top: `${virtualItem.position}px`,
              left: 0,
              width: '100%',
              height: 'auto',
            }}
          >
            <MemoizedPostItem
              post={virtualItem.post}
              onClick={handlePostClick}
            />
          </div>
        ))}
      </div>

      {/* Индикатор загрузки */}
      <div style={{ position: 'relative', zIndex: 1 }}>
        <LoadingIndicator 
          loading={loading}
          allLoaded={allLoaded || totalPosts === 0}
          hasItems={postList.length > 0}
          onVisible={loadMorePosts}
        />
      </div>

      <DebugInfo 
        postCount={postList.length}
        virtualItems={virtualItems}
        loading={loading}
        allLoaded={allLoaded}
        totalHeight={totalHeight}
        scrollMargin={0}
        measuredItems={virtualItems.length}
        totalPosts={totalPosts}
      />
    </div>
  );
});

// Простой throttle
function throttle<T extends (...args: unknown[]) => unknown>(func: T, limit: number): T {
  let inThrottle: boolean;
  return (function (this: unknown, ...args: Parameters<T>) {
    if (!inThrottle) {
      func.apply(this, args);
      inThrottle = true;
      setTimeout(() => (inThrottle = false), limit);
    }
  }) as T;
}

export default PostsThread;
